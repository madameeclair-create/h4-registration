cmake_minimum_required(VERSION 3.22)
project(paper_h4_tools LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(PCL REQUIRED)
find_package(igraph 0.10.6 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(IGRAPH_HEADERS REQUIRED igraph)
find_package(OpenMP REQUIRED)
# 交付包内的MAC源码；不再链接作者电脑上固定路径的静态库。
set(MAC_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/mac/Linux")
include_directories(${PCL_INCLUDE_DIRS} "${MAC_SOURCE}" ${IGRAPH_HEADERS_INCLUDE_DIRS})
link_directories(${PCL_LIBRARY_DIRS})
add_definitions(${PCL_DEFINITIONS})
add_library(mac_algorithm STATIC
 ${MAC_SOURCE}/desc_dec.cpp ${MAC_SOURCE}/funcs.cpp ${MAC_SOURCE}/PCR.cpp
 ${MAC_SOURCE}/registration.cpp ${MAC_SOURCE}/visualization.cpp)
target_compile_options(mac_algorithm PRIVATE -include boost/random.hpp)
target_link_libraries(mac_algorithm PUBLIC ${PCL_LIBRARIES} igraph::igraph OpenMP::OpenMP_CXX)
add_executable(mac_producer producer.cpp PCR_provenance.cpp registration_initialized.cpp)
add_executable(indoor_tools weighted_refine_indoor.cpp)
add_executable(kitti_tools weighted_refine_kitti.cpp)
add_executable(indoor_control control_indoor.cpp)
add_executable(kitti_control control_kitti.cpp)
foreach(t mac_producer indoor_tools kitti_tools indoor_control kitti_control)
 target_compile_options(${t} PRIVATE -include boost/random.hpp)
 target_link_libraries(${t} PRIVATE mac_algorithm)
endforeach()

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(t mac_producer indoor_tools kitti_tools indoor_control kitti_control)
 set_target_properties(${t} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()

option(H4_BUILD_TIMING "Build instrumented timing tools" OFF)
if(H4_BUILD_TIMING)
 add_executable(mac_producer_timed timing/producer_timed.cpp PCR_provenance.cpp timing/registration_timed.cpp)
 add_executable(indoor_tools_timed timing/tools_timed.cpp)
 foreach(t mac_producer_timed indoor_tools_timed)
  target_compile_options(${t} PRIVATE -include boost/random.hpp)
  target_link_libraries(${t} PRIVATE mac_algorithm)
  set_target_properties(${t} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
 endforeach()
endif()
